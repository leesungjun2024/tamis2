CREATE TABLE track_comments (
    id SERIAL PRIMARY KEY,
    track_id UUID REFERENCES tracks(id),
    user_id UUID REFERENCES users(id),
    timestamp_seconds NUMERIC(10, 2) NOT NULL, -- 예: 84.50 (1분 24.5초)
    content TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
대용량 오디오 성능 최적화 팁 (실무 고려사항)
서버사이드 피크 데이터(Peak Data) 사전 생성:

유저가 50MB짜리 긴 WAV 파일을 업로드할 때마다 브라우저가 매번 전체 파일을 로드해서 파형 데이터를 계산하면 CPU가 튀고 로딩이 느려집니다.

백엔드(Node.js / Python 등)에서 오디오 업로드 시점에 미리 파형 피크 데이터(JSON Array)를 미리 추출해 DB나 S3에 저장해두고, 프론트엔드는 이 JSON을 가볍게 받아와 렌더링하면 로딩 속도가 획기적으로 빨라집니다. (Waveurfer.js는 peaks 옵션을 통해 JSON 데이터 주입을 지원합니다.)

오디오 파일 포맷 변환 (Transcoding):

브라우저 호환성과 로딩 속도를 위해 무손실 WAV 파일 외에도 가벼운 MP3 또는 AAC(m4a) 포맷으로 자동 변환(예: FFmpeg 활용)하여 스트리밍용 소스로 함께 제공하는 것이 안전합니다.
